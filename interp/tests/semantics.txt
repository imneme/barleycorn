=== floats: in records, lists, heaps, globals, and recursion
record P
  x: float
  n: int
end
var scale: float = 0.5
func hyp(p: P): float
  return sqrt(p.x * p.x + float(p.n) * float(p.n))
end
func pw(p: P): float
  if p.n == 0 then return 1.0 end
  return p.x * pw(P(p.x, p.n - 1))
end
func main()
  var xs = list[float](8)
  xs.push(2.5); xs.push(-1.25); xs.push(0.5); xs.push(10.0)
  xs.sort()
  for x in xs do
    print x; " ";
  end
  print
  var p = P(1.5, 2)
  print p; " "; min(3.5, 2.25); " "; max(-1.0, -2.0); " "; p.x < 2.0; " "; p.x == 1.5; " "; p.x != 1.5
  var h = heap[float](4)
  h.push(3.0); h.push(1.0); h.push(2.0)
  print h.pop(); " "; h.pop(); " "; hyp(P(3.0, 4)) * scale; " "; pw(P(1.5, 10))
end
--- expect
-1.25 0.5 2.5 10 
P(1.5, 2) 2.25 -1 true true false
1 2 2.5 57.66503906

=== a local read earlier in a loop survives a recursive call later in the loop (and in mutual recursion)
func f(d: int): int
  var acc = 0
  var x = 10 * d + 1
  for i in 0..2 do
    acc += x
    if d > 0 then
      acc += f(d - 1)
    end
  end
  return acc
end
func g(d: int): int
  var acc = 0
  var y = 100 * d + 3
  var w = 0
  while w < 2 do
    acc += y
    if d > 0 then
      acc += h(d - 1)
    end
    w += 1
  end
  return acc
end
func h(d: int): int
  return g(d) + 1
end
func main()
  print f(2); " "; g(2)
end
--- expect
94 848
