=== constant tables: every kind of element, indexed, .len, for, and passed to a function that only reads them
record Step
  risk: word
  at: pos
end

const SCALE = int[0, 2, 4, 5, 7, 9, 11]
const NAMES = str["one", "two", "three", "", "five"]
const KNIGHT = pos[pos(-2, -1), pos(-2, 1), pos(-1, -2), pos(-1, 2), pos(1, -2), pos(1, 2), pos(2, -1), pos(2, 1)]
const STEPS = Step[Step(3, pos(0, 1)), Step(65535, pos(-1, 0))]
const SMALL = byte[1, 2, 255]
const BIG = quad[-9223372036854775807, 0, 1099511627776]
const LONGS = long[-2147483648, 2147483647]
const WORDS = word[0, 65535, 'A']
const YES = bool[true, false, true]

func total(t: int[]): long
  var s: long = 0
  for x in t do s += long(x) end
  return s
end

func first(t: pos[]): int
  return t[0].r * 10 + t[0].c
end

func reads(t: int[]): bool
  return total(t) > 10 and t[1] == 2
end

func main()
  print SCALE.len; " "; SCALE[3]; " "; total(SCALE); " "; reads(SCALE)
  for n in NAMES do print "["; n; "]"; end
  print
  var p = pos(4, 4)
  for d in KNIGHT do print p.r + d.r; ","; p.c + d.c; " "; end
  print
  print STEPS[1].risk; " "; STEPS[0].at.c; " "; STEPS.len; " "; first(KNIGHT); " "; first(dirs4); " "; first(dirs8)
  print SMALL[2]; " "; BIG[0]; " "; BIG[2]; " "; LONGS[0]; " "; LONGS[1]; " "; WORDS[1]; " "; WORDS[2]; " "; YES[1]
  var i: byte = 6

end

=== inkey() is the key held now, key() a new keypress (each typed key is down, then up); ask() starts afresh
func wait(): byte
  var k: byte = 0
  while k == 0 do k = inkey() end
  while inkey() != 0 do end
  return k
